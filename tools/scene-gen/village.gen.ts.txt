const props: any[] = [];
const prop = (id: string, mesh: string, x: number, z: number, o: { rot?: number; y?: number; scale?: number | number[]; col?: string; collision?: string } = {}) => {
  const p: any = { id, mesh, position: [r1(x), r2(o.y ?? 0), r1(z)] };
  if (o.rot) p.rotation = [0, o.rot, 0];
  if (o.scale !== undefined) p.scale = o.scale;
  if (o.collision) p.collision = o.collision;
  props.push(p);
  return p;
};

const RISE = 10 * 0.16;
const zTop = STAIR_Z0 - RUNS * RUN_LEN; // -70.8
const knollBase = N(0, STAIR_Z0 + 3);
const top = knollBase + RUNS * RISE;
const ramp = (z: number) => {
  if (z >= STAIR_Z0) return knollBase;
  if (z >= zTop) return knollBase + ((STAIR_Z0 - z) / (STAIR_Z0 - zTop)) * (RUNS * RISE);
  if (z >= zTop - 12) return top;
  return top - 0.5 * (zTop - 12 - z);
};
const Hs = (x: number, z: number) => {
  const d = Math.abs(x);
  const lateral = d < 6 ? 0 : 0.42 * (d - 6);
  return ramp(z) - lateral;
};
